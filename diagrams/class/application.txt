    - game_id : str
    - player_id : str
    - card_index : int
}

class DiscardCardCommand {
    - game_id : str
    - player_id : str
    - card_index : int
}

class GiveHintCommand {
    - game_id : str
    - from_player : str
    - to_player : str
    - color : Color | None
    - number : Number | None
}

class RegisterCommand {
    - full_name : str
    - email : str
    - username : str
    - password : str
}

class LoginCommand {
    - username : str
    - password : str
}

class CreateLobbyCommand{
    - lobby_id : str
    - max_users : str
    - user_creator : str
}

class JoinLobbyCommand{
    - lobby_id : str
    - user_joined : str
}

    - duration_seconds : int
