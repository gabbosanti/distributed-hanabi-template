@startuml
actor Client

participant "Presentation Layer" as Presentation
participant "Application Layer" as Application
participant "Repository" as Repository


Client -> Presentation : login(username, password)
Presentation -> Application : handleLogin(command)

Application -> Repository : loadUser(username)
Repository --> Application : user / null

alt User exists
Application -> Application : checkPassword()
    alt Authentication succesful
        Application -> Repository : load_player_game_mapping(username)
        Repository --> Application : game / null
            alt Game Exist
                Application -> Repository : loadGame(gameID)
                Repository --> Application : game
                Application -->  Presentation : playerReconnected()
                Presentation --> Client : playerReconnected()
            else Game not found
                Application --> Presentation : loginSuccess()
                Presentation --> Client : loginSuccess()
            end
    else Authentication failed
        Application --> Presentation  : error("Invalid Username/Password")
        Presentation --> Client : error("Invalid Username/Password")
    end
else User not found
    Application --> Presentation : error("Username not found")
    Presentation --> Client : error("Username not found")
end

@enduml
